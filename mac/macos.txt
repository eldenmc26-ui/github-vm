name: MacOS-VNC-Tailscale

on:
  workflow_dispatch:

jobs:
  build:
    runs-on: macos-latest

    steps:
      - name: Install Tailscale
        run: |
          curl -fsSL https://tailscale.com/install.sh | sh

      - name: Connect to Tailscale
        env:
          TAILSCALE_AUTHKEY: ${{ secrets.TAILSCALE_AUTHKEY }}
        run: |
          sudo tailscale up --authkey=$TAILSCALE_AUTHKEY --hostname=github-mac-vnc

      - name: Enable Remote Login (SSH)
        run: |
          sudo systemsetup -setremotelogin on

      - name: Enable VNC (Screen Sharing)
        run: |
          sudo /System/Library/CoreServices/RemoteManagement/ARDAgent.app/Contents/Resources/kickstart \
          -activate -configure -access -on -users runneradmin -privs -all

          sudo /System/Library/CoreServices/RemoteManagement/ARDAgent.app/Contents/Resources/kickstart \
          -restart -agent

      - name: Keep VM alive
        run: |
          echo "Mac ready on Tailscale network"
          sleep 14400
